            if (.@i % 4 == 0) sleep 1;

            if (getmapxy(.@pmap$, .@px, .@py, BL_PC, rid2name(.players[.@i])) == 0)

        }

        for (.@i = 0; .@i < .@n; .@i++) {

            if (.@i % 4 == 0) sleep 1;
            callsub L_Style, .@gids[.@i], 0;
        }
        for (.@i = 0; .@i < getarraysize(.previous_gids); .@i++) {

            if (population_is_shell(.@old) == 0) {
                setd ".stamp_" + .@old, 0;
                setd ".key_" + .@old + "$", "";
            }
        }
        deletearray .previous_gids;
        if (.@n > 0) copyarray .previous_gids, .@gids, .@n;

    // Only style the requesting real player's current map. Ignore client IDs/maps.
    .@req$ = .@atcmd_parameters$[0];

    if (!.ready || .busy) {

    .@now = gettimetick(0);
    if (@dls_refresh_tick && .@now - @dls_refresh_tick < 250) {
        dispbottom "@@reply " + .@req$ + " 1/1 throttled";
        end;

    callsub L_RegisterPlayer;
    .@requester_cid = getcharid(0);
    .@requester_aid = getcharid(3);

L_RegisterPlayer:
    // One world-wide appearance session. Joining friends does not restyle their world.
    // Map-enter retries cannot change the session; temporary player variables reset on logout.
    if (@dls_session_seen) return;
    @dls_session_seen = 1;
    if (.auto_shuffle && getarraysize(.players) == 0) {
        .session_seed = ((.session_seed + rand(1, 9998) - 1) % 9999) + 1;
        .session_epoch++;
    }
    if (inarray(.players, getcharid(3)) < 0)
        .players[getarraysize(.players)] = getcharid(3);
    return;

        if (.garment && (.@kind != 3 || getequipid(EQI_COSTUME_GARMENT, .@cid) <= 0))

    } else {

        if (.upper && getarraysize(.upper_views) > 0 && (.@kind != 3 || getequipid(EQI_COSTUME_HEAD_TOP, .@cid) <= 0)) {
