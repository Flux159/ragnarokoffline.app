    stopnpctimer;

        .busy = 1;
        deletearray .player_maps$;
        for (.@i = getarraysize(.players) - 1; .@i >= 0; .@i--) {
            if (.@i % 4 == 0) sleep 1;
            if (!isloggedin(.players[.@i])) {
                deletearray .players[.@i], 1;
                continue;
            }
            if (getmapxy(.@pmap$, .@px, .@py, BL_PC, rid2name(.players[.@i])) == 0)
                .player_maps$[getarraysize(.player_maps$)] = .@pmap$;
        }
        .@n = getunits(BL_PC, .@gids);
        for (.@i = 0; .@i < .@n; .@i++) {
            // Yield before filters too, even when most units are not eligible.
            if (.@i % 4 == 0) sleep 1;
            callsub L_Style, .@gids[.@i], 0;
        }
        for (.@i = 0; .@i < getarraysize(.previous_gids); .@i++) {
            if (.@i % 4 == 0) sleep 1;
            .@old = .previous_gids[.@i];
            if (population_is_shell(.@old) == 0) {
                setd ".stamp_" + .@old, 0;
                setd ".key_" + .@old + "$", "";
            }
        }
        deletearray .previous_gids;
        if (.@n > 0) copyarray .previous_gids, .@gids, .@n;
        .busy = 0;
    }
    initnpctimer;
    end;

    if (population_is_shell(getcharid(3)) != 0) end;
    if (!.ready || .busy) {
        dispbottom "@@reply " + .@req$ + " 1/1 busy";
        end;
    }
    .@now = gettimetick(0);
    if (@dls_refresh_tick && .@now - @dls_refresh_tick < 250) {
        dispbottom "@@reply " + .@req$ + " 1/1 throttled";
        end;
    }
    @dls_refresh_tick = .@now;
    .@requester_cid = getcharid(0);
    .@requester_aid = getcharid(3);
    if (inarray(.players, .@requester_aid) < 0)
        .players[getarraysize(.players)] = .@requester_aid;
    getmapxy .@map$, .@x, .@y;
    .busy = 1;
    .@n = getmapunits(BL_PC, .@map$, .@gids);
    for (.@i = 0; .@i < .@n; .@i++) {

        callsub L_Style, .@gids[.@i], 1;
    }
    .busy = 0;
    // sleep releases the RID; address the reply explicitly and tolerate logout.
    if (isloggedin(.@requester_aid, .@requester_cid))
        dispbottom "@@reply " + .@req$ + " 1/1 ok", 0xFFFFFF, .@requester_cid;
    end;

OnPCLoginEvent:
    // Remember real players only; AI characters are not tracked.
    if (population_is_shell(getcharid(3)) != 0) end;
    if (inarray(.players, getcharid(3)) < 0)
        .players[getarraysize(.players)] = getcharid(3);
    end;

L_Style:
    .@gid = getarg(0);
    .@force = getarg(1);
    .@kind = population_is_shell(.@gid);
    if (!(.@kind == 3 && .companions) && !(.@kind == 1 && .townspeople) && !(.@kind == 2 && .vendors)) return;
    .@name$ = rid2name(.@gid);
    .@cid = getcharid(0, .@name$);
    if (.@cid <= 0) return;
    if (getmapxy(.@map$, .@x, .@y, BL_PC, .@name$) != 0) return;
    if (.@kind == 1 && !.whole_world && !getmapflag(.@map$, mf_town)) return;
    .@job = readparam(Class, .@cid);
    // A town's guide can switch its theme off; the permanent flag survives restarts.
    .@theme = 0;
    if (.@kind == 1 && .town_themes && !getd("$dls_theme_off_" + .@map$))
        .@theme = getd(".town_theme_" + .@map$);
    .@key$ = .@name$ + ":" + .@kind + ":" + .@job + ":" + .@map$ + ":" + .@theme
        + ":" + getequipid(EQI_COSTUME_HEAD_TOP, .@cid)
        + ":" + getequipid(EQI_COSTUME_HEAD_MID, .@cid)
        + ":" + getequipid(EQI_COSTUME_HEAD_LOW, .@cid)
        + ":" + getequipid(EQI_COSTUME_GARMENT, .@cid);
    .@now = gettimetick(2);
    .@stamp = getd(".stamp_" + .@gid);
    // Unchanged AI are re-applied every 5 s only where a real player is;
    // elsewhere once per change. A player's arrival forces a fresh pass.
    if (!.@force && .@stamp > 0 && getd(".key_" + .@gid + "$") == .@key$
        && (.@now - .@stamp < 5 || inarray(.player_maps$, .@map$) < 0)) return;
    // Avoid overflow and stable across refreshes: no random flickering.
    .@pick = (.@cid % 100000) + .seed * 97;
    // Respect actual costume items, including items occupying multiple head slots.
    // Independent choices per slot; no equipment or inventory is created.
    if (.upper && getarraysize(.upper_views) > 0 && (.@kind != 3 || getequipid(EQI_COSTUME_HEAD_TOP, .@cid) <= 0)) {
        .@view = .upper_views[((.@pick * 37) + 17) % getarraysize(.upper_views)];
        if (.@theme && .theme_upper_count[.@theme] > 0 && ((.@pick * 41) + 7) % 100 < .theme_blend)
            .@view = .theme_upper_views[.theme_upper_start[.@theme] + ((.@pick * 59) + 11) % .theme_upper_count[.@theme]];
        changelook LOOK_HEAD_TOP, .@view, .@cid;
    }
    if (.middle && getarraysize(.middle_views) > 0 && (.@kind != 3 || getequipid(EQI_COSTUME_HEAD_MID, .@cid) <= 0)) {
        .@view = .middle_views[((.@pick * 53) + 29) % getarraysize(.middle_views)];
        if (.@theme && .theme_middle_count[.@theme] > 0 && ((.@pick * 43) + 19) % 100 < .theme_blend)
            .@view = .theme_middle_views[.theme_middle_start[.@theme] + ((.@pick * 67) + 23) % .theme_middle_count[.@theme]];
        changelook LOOK_HEAD_MID, .@view, .@cid;
    }
    if (.lower && getarraysize(.lower_views) > 0 && (.@kind != 3 || getequipid(EQI_COSTUME_HEAD_LOW, .@cid) <= 0)) {
        .@view = .lower_views[((.@pick * 71) + 43) % getarraysize(.lower_views)];
        if (.@theme && .theme_lower_count[.@theme] > 0 && ((.@pick * 47) + 31) % 100 < .theme_blend)
            .@view = .theme_lower_views[.theme_lower_start[.@theme] + ((.@pick * 73) + 37) % .theme_lower_count[.@theme]];
        changelook LOOK_HEAD_BOTTOM, .@view, .@cid;
    }
    if (.garment && getarraysize(.garment_views) > 0 && (.@kind != 3 || getequipid(EQI_COSTUME_GARMENT, .@cid) <= 0)) {
        .@view = .garment_views[((.@pick * 97) + 61) % getarraysize(.garment_views)];
        if (.@theme && .theme_garment_count[.@theme] > 0 && ((.@pick * 61) + 47) % 100 < .theme_blend)
            .@view = .theme_garment_views[.theme_garment_start[.@theme] + ((.@pick * 79) + 53) % .theme_garment_count[.@theme]];
        changelook LOOK_ROBE, .@view, .@cid;
    }
    if (.colours) {
        .@lo = getbattleflag("min_hair_color");
        .@hi = min(8, getbattleflag("max_hair_color"));
        if (.@hi >= .@lo)
            changelook LOOK_HAIR_COLOR, .@lo + ((.@pick / 7) % (.@hi - .@lo + 1)), .@cid;
        // Known conservative first-job palette range; preserve other jobs.
        if (.@job >= Job_Swordman && .@job <= Job_Thief) {
            .@hi = min(3, getbattleflag("max_cloth_color"));
            .@lo = getbattleflag("min_cloth_color");
            if (.@hi >= .@lo)
                changelook LOOK_CLOTHES_COLOR, .@lo + ((.@pick / 11) % (.@hi - .@lo + 1)), .@cid;
        }
    }
    .@alt = 0;
    if (.@job == Job_Rune_Knight || .@job == Job_Rune_Knight2 || .@job == Job_Rune_Knight_T || .@job == Job_Rune_Knight_T2 || .@job == Job_Baby_Rune_Knight || .@job == Job_Baby_Rune_Knight2 || .@job == Job_Dragon_Knight || .@job == Job_Dragon_Knight2)
        .@alt = Job_Rune_Knight_2nd;
    if (.@job == Job_Warlock || .@job == Job_Warlock_T || .@job == Job_Baby_Warlock || .@job == Job_Arch_Mage)
        .@alt = Job_Warlock_2nd;
    if (.@job == Job_Ranger || .@job == Job_Ranger2 || .@job == Job_Ranger_T || .@job == Job_Ranger_T2 || .@job == Job_Baby_Ranger || .@job == Job_Baby_Ranger2 || .@job == Job_Windhawk || .@job == Job_Windhawk2)
        .@alt = Job_Ranger_2nd;
    if (.@job == Job_Arch_Bishop || .@job == Job_Arch_Bishop_T || .@job == Job_Baby_Arch_Bishop || .@job == Job_Cardinal)
        .@alt = Job_ArchBishop_2nd;
    if (.@job == Job_Mechanic || .@job == Job_Mechanic2 || .@job == Job_Mechanic_T || .@job == Job_Mechanic_T2 || .@job == Job_Baby_Mechanic || .@job == Job_Baby_Mechanic2 || .@job == Job_Meister || .@job == Job_Meister2)
        .@alt = Job_Mechanic_2nd;
    if (.@job == Job_Guillotine_Cross || .@job == Job_Guillotine_Cross_T || .@job == Job_Baby_Guillotine_Cross || .@job == Job_Shadow_Cross)
        .@alt = Job_Guillotine_Cross_2nd;
    if (.@job == Job_Royal_Guard || .@job == Job_Royal_Guard2 || .@job == Job_Royal_Guard_T || .@job == Job_Royal_Guard_T2 || .@job == Job_Baby_Royal_Guard || .@job == Job_Baby_Royal_Guard2 || .@job == Job_Imperial_Guard || .@job == Job_Imperial_Guard2)
        .@alt = Job_Royal_Guard_2nd;
    if (.@job == Job_Sorcerer || .@job == Job_Sorcerer_T || .@job == Job_Baby_Sorcerer || .@job == Job_Elemental_Master)
        .@alt = Job_Sorcerer_2nd;
    if (.@job == Job_Minstrel || .@job == Job_Minstrel_T || .@job == Job_Baby_Minstrel || .@job == Job_Troubadour)
        .@alt = Job_Minstrel_2nd;
    if (.@job == Job_Wanderer || .@job == Job_Wanderer_T || .@job == Job_Baby_Wanderer || .@job == Job_Trouvere)
        .@alt = Job_Wanderer_2nd;
    if (.@job == Job_Sura || .@job == Job_Sura_T || .@job == Job_Baby_Sura || .@job == Job_Inquisitor)
        .@alt = Job_Sura_2nd;
    if (.@job == Job_Genetic || .@job == Job_Genetic_T || .@job == Job_Baby_Genetic || .@job == Job_Biolo)
        .@alt = Job_Genetic_2nd;
    if (.@job == Job_Shadow_Chaser || .@job == Job_Shadow_Chaser_T || .@job == Job_Baby_Shadow_Chaser || .@job == Job_Abyss_Chaser)
        .@alt = Job_Shadow_Chaser_2nd;
    if (.bodies && .@alt > 0)
        changelook LOOK_BODY2, ((.@pick / 13) % 2) ? .@alt : .@job, .@cid;

    setd ".stamp_" + .@gid, .@now;
    setd ".key_" + .@gid + "$", .@key$;
    return;
}

// Costume Guide: lets any player switch the current town's theme off or on.
// No GM level needed. The choice is a permanent server variable per town map.
-	script	DaggerTownThemeGuide	-1,{
    .@n$ = "[" + strnpcinfo(1) + "]";
    .@map$ = strnpcinfo(4);
    .@i = inarray(.maps$, .@map$);
    mes .@n$;
    if (.@i < 0) {
        mes "This town keeps the everyday look.";
        close;
    }
    if (!getvariableofnpc(.town_themes, "DaggerLivingStyleV1") || !getvariableofnpc(.townspeople, "DaggerLivingStyleV1")) {
        mes "Town-themed looks are switched off in the Companion Living Style settings.";
        mes "Turn on ^0055AATown-themed looks^000000 and ^0055AAStyle ambient townspeople^000000, then use Apply & restart.";
        close;
    }
    .@off = getd("$dls_theme_off_" + .@map$);
    mes "Townsfolk here mix the ^0055AA" + .labels$[.@i] + "^000000 look into their everyday outfits.";
    mes "Theme in this town: " + (.@off ? "^AA0000off^000000" : "^008800on^000000") + ".";
    next;
    if (select((.@off ? "Turn this town's theme on" : "Turn this town's theme off") + ":Leave") == 1) {
        setd "$dls_theme_off_" + .@map$, !.@off;
        mes .@n$;
        mes "Done. Townsfolk here will change within a few seconds.";
    }
    close;

OnInit:
    setarray .maps$, "prontera", "izlude", "aldebaran", "geffen", "yuno", "lasagna", "payon", "umbala", "hugel", "eclage", "mora", "splendide", "morocc", "veins", "dicastes01", "alberta", "malangdo", "comodo", "dewata", "brasilis", "malaya", "ayothaya", "xmas", "moscovia", "amatsu", "gonryun", "louyang", "einbroch", "einbech", "lighthalzen", "mid_camp", "manuk";

    setarray .labels$, "Royal capital", "Harbour sailors", "Steam and gears", "Arcane scholars", "Arcane scholars", "Forest and hunters", "Forest and hunters", "Forest and hunters", "Forest and hunters", "Forest and hunters", "Forest and hunters", "Forest and hunters", "Desert caravans", "Desert caravans", "Desert caravans", "Harbour sailors", "Harbour sailors", "Seaside summer", "Seaside summer", "Seaside summer", "Seaside summer", "Seaside summer", "Winter festival", "Winter festival", "Eastern islands", "Eastern mountains", "Eastern mountains", "Steam and gears", "Steam and gears", "Steam and gears", "Steam and gears", "Steam and gears";

}
