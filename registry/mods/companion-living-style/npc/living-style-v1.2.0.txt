// Companion Living Style v1.2.0. Appearance packets only: no inventory or stats.
// changelook deliberately does not persist changes to character data.
// Finite scans yield every four units/candidates to stay below script execution limits.
// The client requests a map-local refresh on arrival. A 250 ms discovery timer
// catches newly spawned AI; cached AI get a five-second recovery refresh.
// No promise of zero rendering/network latency or of unavailable client sprites.
-	script	DaggerLivingStyleV1	-1,{
OnInit:
    .ready = 0;
    .busy = 0;
    .mod$ = "companion-living-style";
    bindatcmd "dlsrefresh", strnpcinfo(3) + "::OnMapRefresh", 0, 99;
    .companions = callfunc("F_ModSetting", .mod$, "style_companions", 1);
    .townspeople = callfunc("F_ModSetting", .mod$, "style_townspeople", 1);
    .vendors = callfunc("F_ModSetting", .mod$, "style_vendors", 1);
    .upper = callfunc("F_ModSetting", .mod$, "style_headgear", 1);
    .middle = callfunc("F_ModSetting", .mod$, "style_middle", 1);
    .lower = callfunc("F_ModSetting", .mod$, "style_lower", 1);
    .garment = callfunc("F_ModSetting", .mod$, "style_garment", 1);
    .colours = callfunc("F_ModSetting", .mod$, "style_colours", 1);
    .bodies = callfunc("F_ModSetting", .mod$, "style_bodies", 1);
    .seed = cap_value(callfunc("F_ModSetting", .mod$, "style_seed", 1), 1, 9999);
    // Curated, single-slot candidates from official 1.7.0 item definitions.
    // Read each View from the active database; item IDs are not view IDs.
    // Reject absent or changed definitions, and never mix head and robe views.
    setarray .@upper_ids, 19501, 19506, 19507, 19515, 19516, 19517, 19518, 19519, 19520, 19521, 19522, 19523, 19524, 19525, 19526, 19527, 19529, 19530, 19532, 19533, 19535, 19536, 19537, 19538, 19543, 19544, 19545, 19546, 19547, 19548, 19549, 19557, 19558, 19559, 19560, 19565, 19568, 19569, 19570, 19571, 19572, 19573, 19575, 19576, 19577, 19579, 19581, 19582;
    for (.@i = 0; .@i < getarraysize(.@upper_ids); .@i++) {
        if (.@i % 4 == 0) sleep 1;
        .@id = .@upper_ids[.@i];
        if (getitemname(.@id) == "null") continue;
        if (getiteminfo(.@id, ITEMINFO_LOCATIONS) != EQP_COSTUME_HEAD_TOP) continue;
        .@view = getiteminfo(.@id, ITEMINFO_VIEW);
        if (.@view > 0 && inarray(.upper_views, .@view) < 0)
            .upper_views[getarraysize(.upper_views)] = .@view;
    }
    setarray .@middle_ids, 19504, 19509, 19510, 19511, 19512, 19542, 19550, 19551, 19583, 19609, 19611, 19621, 19624, 19644, 19669, 19722, 19734, 19735, 19742, 19752, 19755, 19765, 19769, 19776, 19787, 19788, 19810, 19826, 19846, 19853, 19871, 19877;
    for (.@i = 0; .@i < getarraysize(.@middle_ids); .@i++) {
        if (.@i % 4 == 0) sleep 1;
        .@id = .@middle_ids[.@i];
        if (getitemname(.@id) == "null") continue;
        if (getiteminfo(.@id, ITEMINFO_LOCATIONS) != EQP_COSTUME_HEAD_MID) continue;
        .@view = getiteminfo(.@id, ITEMINFO_VIEW);
        if (.@view > 0 && inarray(.middle_views, .@view) < 0)
            .middle_views[getarraysize(.middle_views)] = .@view;
    }
    setarray .@lower_ids, 19505, 19513, 19514, 19528, 19541, 19552, 19553, 19566, 19584, 19605, 19606, 19634, 19636, 19651, 19672, 19683, 19692, 19713, 19726, 19764, 19770, 19783, 19785, 19805, 19817, 19947, 19952, 19992, 19993, 20015, 20022, 20029;
    for (.@i = 0; .@i < getarraysize(.@lower_ids); .@i++) {
        if (.@i % 4 == 0) sleep 1;
        .@id = .@lower_ids[.@i];
        if (getitemname(.@id) == "null") continue;
        if (getiteminfo(.@id, ITEMINFO_LOCATIONS) != EQP_COSTUME_HEAD_LOW) continue;
        .@view = getiteminfo(.@id, ITEMINFO_VIEW);
        if (.@view > 0 && inarray(.lower_views, .@view) < 0)
            .lower_views[getarraysize(.lower_views)] = .@view;
    }
    setarray .@garment_ids, 20500, 20501, 20502, 20503, 20504, 20505, 20507, 20509, 20510, 20511, 20512, 20513, 20514, 20516, 20517, 20518, 20521, 20523, 20525, 20526, 20527, 20528, 20530, 20533;
    for (.@i = 0; .@i < getarraysize(.@garment_ids); .@i++) {
        if (.@i % 4 == 0) sleep 1;
        .@id = .@garment_ids[.@i];
        if (getitemname(.@id) == "null") continue;
        if (getiteminfo(.@id, ITEMINFO_LOCATIONS) != EQP_COSTUME_GARMENT) continue;
        .@view = getiteminfo(.@id, ITEMINFO_VIEW);
        if (.@view > 0 && inarray(.garment_views, .@view) < 0)
            .garment_views[getarraysize(.garment_views)] = .@view;
    }
    .ready = 1;
    initnpctimer;
    end;

OnTimer250:
    stopnpctimer;
    if (.ready && !.busy && (.companions || .townspeople || .vendors)) {
        .busy = 1;
        .@n = getunits(BL_PC, .@gids);
        for (.@i = 0; .@i < .@n; .@i++) {
            // Yield before filters too, even when most units are not eligible.
            if (.@i % 4 == 0) sleep 1;
            callsub L_Style, .@gids[.@i], 0;
        }
        for (.@i = 0; .@i < getarraysize(.previous_gids); .@i++) {
            if (.@i % 4 == 0) sleep 1;
            .@old = .previous_gids[.@i];
            if (population_is_shell(.@old) == 0) {
                setd ".stamp_" + .@old, 0;
                setd ".key_" + .@old + "$", "";
            }
        }
        deletearray .previous_gids;
        if (.@n > 0) copyarray .previous_gids, .@gids, .@n;
        .busy = 0;
    }
    initnpctimer;
    end;

OnMapRefresh:
    // Only style the requesting real player's current map. Ignore client IDs/maps.
    .@req$ = .@atcmd_parameters$[0];
    if (population_is_shell(getcharid(3)) != 0) end;
    if (!.ready || .busy) {
        dispbottom "@@reply " + .@req$ + " 1/1 busy";
        end;
    }
    .@now = gettimetick(0);
    if (@dls_refresh_tick && .@now - @dls_refresh_tick < 250) {
        dispbottom "@@reply " + .@req$ + " 1/1 throttled";
        end;
    }
    @dls_refresh_tick = .@now;
    .@requester_cid = getcharid(0);
    .@requester_aid = getcharid(3);
    getmapxy .@map$, .@x, .@y;
    .busy = 1;
    .@n = getmapunits(BL_PC, .@map$, .@gids);
    for (.@i = 0; .@i < .@n; .@i++) {
        if (.@i % 4 == 0) sleep 1;
        callsub L_Style, .@gids[.@i], 1;
    }
    .busy = 0;
    // sleep releases the RID; address the reply explicitly and tolerate logout.
    if (isloggedin(.@requester_aid, .@requester_cid))
        dispbottom "@@reply " + .@req$ + " 1/1 ok", 0xFFFFFF, .@requester_cid;
    end;

L_Style:
    .@gid = getarg(0);
    .@force = getarg(1);
    .@kind = population_is_shell(.@gid);
    if (!(.@kind == 3 && .companions) && !(.@kind == 1 && .townspeople) && !(.@kind == 2 && .vendors)) return;
    .@name$ = rid2name(.@gid);
    .@cid = getcharid(0, .@name$);
    if (.@cid <= 0) return;
    if (getmapxy(.@map$, .@x, .@y, BL_PC, .@name$) != 0) return;
    if (.@kind == 1 && !getmapflag(.@map$, mf_town)) return;
    .@job = readparam(Class, .@cid);
    .@key$ = .@name$ + ":" + .@kind + ":" + .@job + ":" + .@map$
        + ":" + getequipid(EQI_COSTUME_HEAD_TOP, .@cid)
        + ":" + getequipid(EQI_COSTUME_HEAD_MID, .@cid)
        + ":" + getequipid(EQI_COSTUME_HEAD_LOW, .@cid)
        + ":" + getequipid(EQI_COSTUME_GARMENT, .@cid);
    .@now = gettimetick(2);
    .@stamp = getd(".stamp_" + .@gid);
    if (!.@force && .@stamp > 0 && .@now - .@stamp < 5 && getd(".key_" + .@gid + "$") == .@key$) return;
    // Avoid overflow and stable across refreshes: no random flickering.
    .@pick = (.@cid % 100000) + .seed * 97;
    // Respect actual costume items, including items occupying multiple head slots.
    // Independent choices per slot; no equipment or inventory is created.
    if (.upper && getarraysize(.upper_views) > 0 && (.@kind != 3 || getequipid(EQI_COSTUME_HEAD_TOP, .@cid) <= 0))
        changelook LOOK_HEAD_TOP, .upper_views[((.@pick * 37) + 17) % getarraysize(.upper_views)], .@cid;
    if (.middle && getarraysize(.middle_views) > 0 && (.@kind != 3 || getequipid(EQI_COSTUME_HEAD_MID, .@cid) <= 0))
        changelook LOOK_HEAD_MID, .middle_views[((.@pick * 53) + 29) % getarraysize(.middle_views)], .@cid;
    if (.lower && getarraysize(.lower_views) > 0 && (.@kind != 3 || getequipid(EQI_COSTUME_HEAD_LOW, .@cid) <= 0))
        changelook LOOK_HEAD_BOTTOM, .lower_views[((.@pick * 71) + 43) % getarraysize(.lower_views)], .@cid;
    if (.garment && getarraysize(.garment_views) > 0 && (.@kind != 3 || getequipid(EQI_COSTUME_GARMENT, .@cid) <= 0))
        changelook LOOK_ROBE, .garment_views[((.@pick * 97) + 61) % getarraysize(.garment_views)], .@cid;
    if (.colours) {
        .@lo = getbattleflag("min_hair_color");
        .@hi = min(8, getbattleflag("max_hair_color"));
        if (.@hi >= .@lo)
            changelook LOOK_HAIR_COLOR, .@lo + ((.@pick / 7) % (.@hi - .@lo + 1)), .@cid;
        // Known conservative first-job palette range; preserve other jobs.
        if (.@job >= Job_Swordman && .@job <= Job_Thief) {
            .@hi = min(3, getbattleflag("max_cloth_color"));
            .@lo = getbattleflag("min_cloth_color");
            if (.@hi >= .@lo)
                changelook LOOK_CLOTHES_COLOR, .@lo + ((.@pick / 11) % (.@hi - .@lo + 1)), .@cid;
        }
    }
    .@alt = 0;
    if (.@job == Job_Rune_Knight || .@job == Job_Rune_Knight2 || .@job == Job_Rune_Knight_T || .@job == Job_Rune_Knight_T2 || .@job == Job_Baby_Rune_Knight || .@job == Job_Baby_Rune_Knight2 || .@job == Job_Dragon_Knight || .@job == Job_Dragon_Knight2)
        .@alt = Job_Rune_Knight_2nd;
    if (.@job == Job_Warlock || .@job == Job_Warlock_T || .@job == Job_Baby_Warlock || .@job == Job_Arch_Mage)
        .@alt = Job_Warlock_2nd;
    if (.@job == Job_Ranger || .@job == Job_Ranger2 || .@job == Job_Ranger_T || .@job == Job_Ranger_T2 || .@job == Job_Baby_Ranger || .@job == Job_Baby_Ranger2 || .@job == Job_Windhawk || .@job == Job_Windhawk2)
        .@alt = Job_Ranger_2nd;
    if (.@job == Job_Arch_Bishop || .@job == Job_Arch_Bishop_T || .@job == Job_Baby_Arch_Bishop || .@job == Job_Cardinal)
        .@alt = Job_ArchBishop_2nd;
    if (.@job == Job_Mechanic || .@job == Job_Mechanic2 || .@job == Job_Mechanic_T || .@job == Job_Mechanic_T2 || .@job == Job_Baby_Mechanic || .@job == Job_Baby_Mechanic2 || .@job == Job_Meister || .@job == Job_Meister2)
        .@alt = Job_Mechanic_2nd;
    if (.@job == Job_Guillotine_Cross || .@job == Job_Guillotine_Cross_T || .@job == Job_Baby_Guillotine_Cross || .@job == Job_Shadow_Cross)
        .@alt = Job_Guillotine_Cross_2nd;
    if (.@job == Job_Royal_Guard || .@job == Job_Royal_Guard2 || .@job == Job_Royal_Guard_T || .@job == Job_Royal_Guard_T2 || .@job == Job_Baby_Royal_Guard || .@job == Job_Baby_Royal_Guard2 || .@job == Job_Imperial_Guard || .@job == Job_Imperial_Guard2)
        .@alt = Job_Royal_Guard_2nd;
    if (.@job == Job_Sorcerer || .@job == Job_Sorcerer_T || .@job == Job_Baby_Sorcerer || .@job == Job_Elemental_Master)
        .@alt = Job_Sorcerer_2nd;
    if (.@job == Job_Minstrel || .@job == Job_Minstrel_T || .@job == Job_Baby_Minstrel || .@job == Job_Troubadour)
        .@alt = Job_Minstrel_2nd;
    if (.@job == Job_Wanderer || .@job == Job_Wanderer_T || .@job == Job_Baby_Wanderer || .@job == Job_Trouvere)
        .@alt = Job_Wanderer_2nd;
    if (.@job == Job_Sura || .@job == Job_Sura_T || .@job == Job_Baby_Sura || .@job == Job_Inquisitor)
        .@alt = Job_Sura_2nd;
    if (.@job == Job_Genetic || .@job == Job_Genetic_T || .@job == Job_Baby_Genetic || .@job == Job_Biolo)
        .@alt = Job_Genetic_2nd;
    if (.@job == Job_Shadow_Chaser || .@job == Job_Shadow_Chaser_T || .@job == Job_Baby_Shadow_Chaser || .@job == Job_Abyss_Chaser)
        .@alt = Job_Shadow_Chaser_2nd;
    if (.bodies && .@alt > 0)
        changelook LOOK_BODY2, ((.@pick / 13) % 2) ? .@alt : .@job, .@cid;

    setd ".stamp_" + .@gid, .@now;
    setd ".key_" + .@gid + "$", .@key$;
    return;
}
