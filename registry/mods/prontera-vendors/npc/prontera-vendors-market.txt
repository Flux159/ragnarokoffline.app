//===== Ragnarok Offline: prontera-vendors =================================
//= The dynamic market. Prices follow the market, then drift back to the
//= price list:
//=
//= - Trades move them. Once a minute this reads the server's own trade log
//=   (picklog, types V and B): a player buying from a stall pushes a price
//=   up, selling into a buying store pushes it down, and a player's own stall
//=   or store trading with the customers nudges it a little. How far depends
//=   on how much of the item changes hands on a normal day ($@pv_vol, from
//=   prontera-vendors-market-data.txt).
//= - A move spills over to the item's group, by the group's share.
//= - Market news, now and then, moves a group for a few days ("Market
//=   news"). With "Announce market news" on (the default) the server also
//=   announces when one starts and when it ends; off, only the board shows
//=   it.
//= - Every ten minutes each move fades with the recovery half-life; the time
//=   the server was off is faded in one step at start.
//= - War of Emperium: when WoE starts (agitcheck, agitcheck2, agitcheck3),
//=   its supplies (the woe_season news' potions, gems and bottles) move up
//=   10-20 % once and fade like any move; announced with "Announce market
//=   news" on.
//= - With the hunted supply on, its scarcity factor ($@pvs_pct, from
//=   prontera-vendors-supply.txt) is multiplied in as well.
//=
//= The index lives in this script's own permanent variables ($pv_*). What it
//= publishes is $@pop_item_pct[<item id>]: the population engine prices this
//= mod's stalls and buyers and its customers' market price by it. The board
//= and @market read the same.
//=
//= Off unless "Dynamic market" is on in the mod's settings.
//===========================================================================
-	script	ProntVendorsMarket	-1,{
	end;

OnInit:
	.mod$ = "prontera-vendors";
	.on = callfunc("F_ModSetting", .mod$, "dynamic_market", 0);
	.strength = callfunc("F_ModSetting", .mod$, "market_strength", 100);
	.halflife = callfunc("F_ModSetting", .mod$, "market_recovery_hours", 72) * 3600;
	// The news events, and whether the server announces them (and WoE's
	// moves) or leaves them to the board.
	.news = callfunc("F_ModSetting", .mod$, "market_news", 1);
	.announce = callfunc("F_ModSetting", .mod$, "market_news_announce", 1);
	.ready = 0;
	.passes = 0;
	$@pv_board_on = .on;
	bindatcmd "market", strnpcinfo(3) + "::OnAtMarket", 99, 99;
	if (.on)
		initnpctimer;
	end;

// A few seconds after start, so the data script has filled $@pv_* first.
OnTimer5000:
	if (!.ready)
		callsub S_Start;
	end;

OnTimer60000:
	callsub S_Pass;
	initnpctimer;
	end;

// ---------------------------------------------------------------------------
// Start: the index back from its variables, the downtime faded, the news
// matched to this build's events.
S_Start:
	freeloop(1);
	deletearray $@pv_slot;
	for (.@i = 0; .@i < $pv_mk_n; .@i++)
		$@pv_slot[$pv_mk_id[.@i]] = .@i + 1;
	.@now = gettimetick(2);
	if ($pv_clock > 0 && .@now > $pv_clock)
		callsub S_FadeAll, .@now - $pv_clock;
	$pv_clock = .@now;
	if ($pv_lastlog <= 0) {
		// Never run before: start from now, not from the server's whole history.
		query_logsql("SELECT COALESCE(MAX(id), 0) FROM picklog", .@max);
		$pv_lastlog = .@max;
	}
	.ready = 1;
	callsub S_Publish;
	freeloop(0);
	return;

// ---------------------------------------------------------------------------
// Once a minute: new trades, news now and then, fading every ten minutes,
// and the prices and board published.
S_Pass:
	if (!.ready)
		callsub S_Start;
	freeloop(1);
	.@n = query_logsql("SELECT p.id, p.type, p.nameid, p.amount, (c.char_id IS NOT NULL) FROM picklog p LEFT JOIN `char` c ON c.char_id = p.char_id WHERE p.id > " + $pv_lastlog + " AND p.type IN ('V','B') ORDER BY p.id LIMIT 500", .@id, .@type$, .@nameid, .@amount, .@real);
	for (.@i = 0; .@i < .@n; .@i++) {
		$pv_lastlog = .@id[.@i];
		if (!.@real[.@i] || .@amount[.@i] == 0)
			continue; // a fake player's side of a trade
		// Per mille of a full push: buying from a stall +, selling into a buying
		// store -, a player's own stall or store trading with customers a little.
		if (.@type$[.@i] == "V")
			.@dir = .@amount[.@i] > 0 ? 1000 : 300;
		else
			.@dir = .@amount[.@i] < 0 ? -1000 : -300;
		callsub S_Trade, .@nameid[.@i], .@amount[.@i] < 0 ? -.@amount[.@i] : .@amount[.@i], .@dir;
	}
	.passes++;
	if (.passes % 10 == 0) {
		.@now = gettimetick(2);
		callsub S_FadeAll, .@now - $pv_clock;
		$pv_clock = .@now;
	}
	// War of Emperium: its supplies go up once as it starts ($pv_woe keeps a
	// restart during WoE from doing it again).
	.@woe = agitcheck() || agitcheck2() || agitcheck3();
	if (.@woe && !$pv_woe) {
		$pv_woe = 1;
		callsub S_Woe;
	} else if (!.@woe && $pv_woe) {
		$pv_woe = 0;
		if (.announce)
			announce "[Prontera Market] The War of Emperium is over; potions and gems settle.", bc_all;
	}
	// News: about once a week, at most two at once, none again within four weeks.
	if (.news && rand(10080) == 0)
		callsub S_StartNews, "";
	callsub S_Publish;
	freeloop(0);
	return;

// ---------------------------------------------------------------------------
// S_Trade <item id>, <amount>, <direction per mille>: move the item by its
// share of a normal day's volume, and its group by the group's share.
// At strength 100 a full day's volume moves a price by about 40 %.
S_Trade:
	.@item = getarg(0);
	.@vol = $@pv_vol[.@item] > 0 ? $@pv_vol[.@item] : 8;
	.@chg = .strength * 4 * getarg(1) * getarg(2) / .@vol; // per million
	if (.@chg > 250000) .@chg = 250000;
	if (.@chg < -250000) .@chg = -250000;
	if (.@chg == 0)
		return;
	callsub S_Move, .@item, .@chg;
	.@g = $@pv_ig[.@item];
	if (.@g > 0) {
		.@g--;
		.@part = .@chg * $@pv_gshare[.@g] / 100;
		for (.@j = 0; .@j < $@pv_glen[.@g]; .@j++) {
			.@other = $@pv_gitem[$@pv_gstart[.@g] + .@j];
			if (.@other != .@item)
				callsub S_Move, .@other, .@part;
		}
	}
	return;

// S_Move <item id>, <change per million>: the item's index (1000000 = the
// price list), kept between half and twice.
S_Move:
	.@item = getarg(0);
	.@s = $@pv_slot[.@item];
	if (!.@s) {
		.@s = $pv_mk_n + 1;
		$pv_mk_id[.@s - 1] = .@item;
		$pv_mk_v[.@s - 1] = 1000000;
		$pv_mk_n = .@s;
		$@pv_slot[.@item] = .@s;
	}
	.@v = $pv_mk_v[.@s - 1] * (1000000 + getarg(1)) / 1000000;
	if (.@v < 500000) .@v = 500000;
	if (.@v > 2000000) .@v = 2000000;
	$pv_mk_v[.@s - 1] = .@v;
	$pv_mk_t[.@s - 1] = gettimetick(2);
	return;

// S_Fade <value>, <seconds>: what is left of a move after that long, halving
// every half-life (whole halvings exact, the rest within a few percent).
S_Fade:
	.@v = getarg(0);
	.@h = getarg(1) * 1000 / .halflife; // in thousandths of a half-life
	if (.@h >= 40000)
		return 0;
	for (.@k = 0; .@k < .@h / 1000; .@k++)
		.@v /= 2;
	return .@v * (1000000 - (.@h % 1000) * 693) / 1000000;

// S_FadeAll <seconds>: fade every move; drop those back within 0.5 %.
S_FadeAll:
	.@secs = getarg(0);
	.@keep = 0;
	for (.@i = 0; .@i < $pv_mk_n; .@i++) {
		.@dev = callsub(S_Fade, $pv_mk_v[.@i] - 1000000, .@secs);
		.@item = $pv_mk_id[.@i];
		if (.@dev > -5000 && .@dev < 5000) {
			$@pv_slot[.@item] = 0;
			continue;
		}
		$pv_mk_id[.@keep] = .@item;
		$pv_mk_v[.@keep] = 1000000 + .@dev;
		$pv_mk_t[.@keep] = $pv_mk_t[.@i];
		$@pv_slot[.@item] = .@keep + 1;
		.@keep++;
	}
	for (.@i = .@keep; .@i < $pv_mk_n; .@i++) {
		$pv_mk_id[.@i] = 0;
		$pv_mk_v[.@i] = 0;
		$pv_mk_t[.@i] = 0;
	}
	$pv_mk_n = .@keep;
	return;

// S_Woe: the woe_season news' items up 10-20 % each, as a trade would move
// them, without starting the news itself.
S_Woe:
	for (.@e = 0; .@e < $@pv_evcount; .@e++) {
		if ($@pv_evkey$[.@e] != "woe_season")
			continue;
		for (.@f = 0; .@f < $@pv_fxcount; .@f++) {
			if ($@pv_fxev[.@f] != .@e)
				continue;
			for (.@j = 0; .@j < $@pv_fxlen[.@f]; .@j++)
				callsub S_Move, $@pv_fxitem[$@pv_fxstart[.@f] + .@j], rand(100000, 200000);
		}
	}
	if (.announce)
		announce "[Prontera Market] The War of Emperium has begun: guilds buy up potions and gems.", bc_all;
	return;

// ---------------------------------------------------------------------------
// S_StartNews "<event key>" (or "" for a random one): start it now, its
// changes rolled. Returns 1 if it started.
S_StartNews:
	.@now = gettimetick(2);
	.@running = 0;
	for (.@k = 0; .@k < $pv_nw_n; .@k++)
		if (.@now < $pv_nw_end[.@k])
			.@running++;
	.@want$ = getarg(0);
	deletearray .@free;
	for (.@e = 0; .@e < $@pv_evcount; .@e++) {
		.@key$ = $@pv_evkey$[.@e];
		if (.@want$ != "" && .@key$ != .@want$)
			continue;
		.@busy = 0;
		for (.@k = 0; .@k < $pv_nw_n; .@k++)
			if ($pv_nw_key$[.@k] == .@key$ && .@now < $pv_nw_end[.@k])
				.@busy = 1;
		if (.@busy)
			continue;
		if (.@want$ == "" && .@now - getd("$pv_nwl_" + .@key$) < 28 * 86400)
			continue;
		.@free[getarraysize(.@free)] = .@e;
	}
	if (!getarraysize(.@free) || (.@want$ == "" && .@running >= 2))
		return 0;
	.@e = .@free[rand(getarraysize(.@free))];
	// Up to two effects per event: roll each in its range.
	.@c0 = 0; .@c1 = 0; .@n = 0;
	for (.@f = 0; .@f < $@pv_fxcount; .@f++) {
		if ($@pv_fxev[.@f] != .@e)
			continue;
		.@c = rand($@pv_fxmin[.@f], $@pv_fxmax[.@f]);
		if (.@n == 0) .@c0 = .@c; else .@c1 = .@c;
		.@n++;
	}
	// One slot per event: a faded one is reused.
	for (.@k = 0; .@k < $pv_nw_n; .@k++)
		if ($pv_nw_key$[.@k] == $@pv_evkey$[.@e])
			break;
	if (.@k == $pv_nw_n)
		$pv_nw_n++;
	$pv_nw_key$[.@k] = $@pv_evkey$[.@e];
	$pv_nw_end[.@k] = .@now + $@pv_evdays[.@e] * 86400;
	$pv_nw_c0[.@k] = .@c0;
	$pv_nw_c1[.@k] = .@c1;
	$pv_nw_told[.@k] = 0;
	setd "$pv_nwl_" + $@pv_evkey$[.@e], .@now;
	if (.announce)
		announce "[Prontera Market] " + $@pv_evtext$[.@e], bc_all;
	return 1;

// ---------------------------------------------------------------------------
// S_Publish: each item's price percentage for the engine (trades times
// news), the board's lines, and news that has faded dropped.
S_Publish:
	freeloop(1);
	for (.@i = 0; .@i < getarraysize($@pv_pub); .@i++)
		$@pop_item_pct[$@pv_pub[.@i]] = 0;
	deletearray $@pv_pub;
	.@now = gettimetick(2);
	deletearray .@evitem;
	deletearray .@evdev;
	// News: its share of each item's price (per million), full while it
	// runs, then fading like any move.
	.@keep = 0;
	for (.@k = 0; .@k < $pv_nw_n; .@k++) {
		.@w = 1000000;
		if (.@now >= $pv_nw_end[.@k])
			.@w = callsub(S_Fade, 1000000, .@now - $pv_nw_end[.@k]);
		if (.@w < 20000)
			continue; // faded away
		if (.@keep != .@k) {
			$pv_nw_key$[.@keep] = $pv_nw_key$[.@k];
			$pv_nw_end[.@keep] = $pv_nw_end[.@k];
			$pv_nw_c0[.@keep] = $pv_nw_c0[.@k];
			$pv_nw_c1[.@keep] = $pv_nw_c1[.@k];
			$pv_nw_told[.@keep] = $pv_nw_told[.@k];
		}
		.@e = -1;
		for (.@x = 0; .@x < $@pv_evcount; .@x++)
			if ($@pv_evkey$[.@x] == $pv_nw_key$[.@keep])
				.@e = .@x;
		// Its end, announced once: within the hour it ended, so news that
		// ran out while the server was off isn't announced late.
		if (.@now >= $pv_nw_end[.@keep] && !$pv_nw_told[.@keep]) {
			$pv_nw_told[.@keep] = 1;
			if (.announce && .@e >= 0 && .@now - $pv_nw_end[.@keep] < 3600)
				announce "[Prontera Market] " + $@pv_evend$[.@e], bc_all;
		}
		.@n = 0;
		for (.@f = 0; .@e >= 0 && .@f < $@pv_fxcount; .@f++) {
			if ($@pv_fxev[.@f] != .@e)
				continue;
			.@c = .@n == 0 ? $pv_nw_c0[.@keep] : $pv_nw_c1[.@keep];
			.@n++;
			for (.@j = 0; .@j < $@pv_fxlen[.@f]; .@j++) {
				.@item = $@pv_fxitem[$@pv_fxstart[.@f] + .@j];
				for (.@y = 0; .@y < getarraysize(.@evitem); .@y++)
					if (.@evitem[.@y] == .@item)
						break;
				.@evitem[.@y] = .@item;
				.@evdev[.@y] += .@c * .@w / 100;
			}
		}
		.@keep++;
	}
	for (.@k = .@keep; .@k < $pv_nw_n; .@k++) {
		$pv_nw_key$[.@k] = "";
		$pv_nw_end[.@k] = 0;
		$pv_nw_c0[.@k] = 0;
		$pv_nw_c1[.@k] = 0;
		$pv_nw_told[.@k] = 0;
	}
	$pv_nw_n = .@keep;

	// Every item that moved: trades times news, as a percentage.
	deletearray .@pid;
	deletearray .@ppct;
	for (.@i = 0; .@i < $pv_mk_n; .@i++) {
		.@pid[.@i] = $pv_mk_id[.@i];
		.@pv[.@i] = $pv_mk_v[.@i];
	}
	.@np = $pv_mk_n;
	for (.@y = 0; .@y < getarraysize(.@evitem); .@y++) {
		.@s = $@pv_slot[.@evitem[.@y]];
		if (.@s > 0) {
			.@pv[.@s - 1] = .@pv[.@s - 1] * (1000000 + .@evdev[.@y]) / 1000000;
		} else {
			.@pid[.@np] = .@evitem[.@y];
			.@pv[.@np] = 1000000 + .@evdev[.@y];
			.@np++;
		}
	}
	// The hunted supply's scarcity (prontera-vendors-supply.txt sets
	// $@pvs_pct), on top of trades and news.
	if (getarraysize($@pvs_plist) > 0) {
		for (.@i = 0; .@i < .@np; .@i++)
			.@pos[.@pid[.@i]] = .@i + 1;
		for (.@j = 0; .@j < getarraysize($@pvs_plist); .@j++) {
			.@item = $@pvs_plist[.@j];
			if (.@pos[.@item] > 0) {
				.@pv[.@pos[.@item] - 1] = .@pv[.@pos[.@item] - 1] * $@pvs_pct[.@item] / 100;
			} else {
				.@pid[.@np] = .@item;
				.@pv[.@np] = $@pvs_pct[.@item] * 10000;
				.@np++;
			}
		}
	}
	for (.@i = 0; .@i < .@np; .@i++) {
		.@pct = (.@pv[.@i] + 5000) / 10000;
		if (.@pct < 50) .@pct = 50;
		if (.@pct > 200) .@pct = 200;
		.@ppct[.@i] = .@pct;
		if (.@pct != 100) {
			$@pop_item_pct[.@pid[.@i]] = .@pct;
			$@pv_pub[getarraysize($@pv_pub)] = .@pid[.@i];
		}
	}

	// The board: the five rising and falling most, and the news.
	deletearray $@pv_board_up$;
	deletearray $@pv_board_down$;
	deletearray $@pv_board_news$;
	for (.@r = 0; .@r < 5; .@r++) {
		.@best = -1; .@worst = -1;
		for (.@i = 0; .@i < .@np; .@i++) {
			if (.@ppct[.@i] > 100 && (.@best < 0 || .@ppct[.@i] > .@ppct[.@best])) .@best = .@i;
			if (.@ppct[.@i] < 100 && .@ppct[.@i] > 0 && (.@worst < 0 || .@ppct[.@i] < .@ppct[.@worst])) .@worst = .@i;
		}
		if (.@best >= 0) {
			$@pv_board_up$[.@r] = getitemname(.@pid[.@best]) + " +" + (.@ppct[.@best] - 100) + "%";
			.@ppct[.@best] = 100;
		}
		if (.@worst >= 0) {
			$@pv_board_down$[.@r] = getitemname(.@pid[.@worst]) + " -" + (100 - .@ppct[.@worst]) + "%";
			.@ppct[.@worst] = 100;
		}
	}
	for (.@k = 0; .@k < $pv_nw_n; .@k++) {
		for (.@x = 0; .@x < $@pv_evcount; .@x++) {
			if ($@pv_evkey$[.@x] != $pv_nw_key$[.@k])
				continue;
			if (.@now < $pv_nw_end[.@k]) {
				.@days = ($pv_nw_end[.@k] - .@now + 86399) / 86400;
				$@pv_board_news$[getarraysize($@pv_board_news$)] = $@pv_evtext$[.@x] + " (" + .@days + " more day" + (.@days > 1 ? "s" : "") + ")";
			} else {
				$@pv_board_news$[getarraysize($@pv_board_news$)] = $@pv_evtext$[.@x] + " (fading)";
			}
		}
	}
	freeloop(0);
	return;

// ---------------------------------------------------------------------------
// @market [<item> | news [<event>] | reset]  (GMs)
OnAtMarket:
	if (!.on) {
		dispbottom "The dynamic market is off: Settings > Mods > prontera-vendors > Dynamic market.";
		end;
	}
	if (!.ready)
		callsub S_Start;
	.@arg$ = .@atcmd_numparameters > 0 ? .@atcmd_parameters$[0] : "";
	if (.@arg$ == "reset") {
		for (.@i = 0; .@i < $pv_mk_n; .@i++) {
			$@pv_slot[$pv_mk_id[.@i]] = 0;
			$pv_mk_id[.@i] = 0;
			$pv_mk_v[.@i] = 0;
			$pv_mk_t[.@i] = 0;
		}
		$pv_mk_n = 0;
		for (.@k = 0; .@k < $pv_nw_n; .@k++) {
			$pv_nw_end[.@k] = 0;
			$pv_nw_told[.@k] = 1;
		}
		$pv_nw_n = 0;
		callsub S_Publish;
		dispbottom "Every price is back at the price list, and the news is cleared.";
		end;
	}
	if (.@arg$ == "news") {
		if (.@atcmd_numparameters > 1) {
			if (callsub(S_StartNews, .@atcmd_parameters$[1])) {
				callsub S_Publish;
				dispbottom "Started.";
			} else {
				dispbottom "No such event, or it is running already.";
			}
			end;
		}
		.@now = gettimetick(2);
		for (.@e = 0; .@e < $@pv_evcount; .@e++) {
			.@state$ = "";
			for (.@k = 0; .@k < $pv_nw_n; .@k++)
				if ($pv_nw_key$[.@k] == $@pv_evkey$[.@e])
					.@state$ = .@now < $pv_nw_end[.@k] ? " [running]" : " [fading]";
			dispbottom "  " + $@pv_evkey$[.@e] + " (" + $@pv_evdays[.@e] + " days)" + .@state$;
		}
		dispbottom "@market news <event> starts one now.";
		end;
	}
	if (.@arg$ != "") {
		.@item = atoi(.@arg$);
		if (.@item <= 0) {
			.@found = searchitem(.@ids, .@arg$);
			if (.@found <= 0) {
				dispbottom "No such item (use its name or id).";
				end;
			}
			.@item = .@ids[0];
		}
		.@pct = $@pop_item_pct[.@item] > 0 ? $@pop_item_pct[.@item] : 100;
		.@s = $@pv_slot[.@item];
		.@trade = .@s > 0 ? ($pv_mk_v[.@s - 1] - 1000000) / 10000 : 0;
		dispbottom getitemname(.@item) + ": " + .@pct + "% of the price list (trades " + (.@trade >= 0 ? "+" : "") + .@trade + "%, the rest news and supply), a normal day " + ($@pv_vol[.@item] > 0 ? $@pv_vol[.@item] : 8) + " traded.";
		if (.@s > 0)
			dispbottom "  last moved " + ((gettimetick(2) - $pv_mk_t[.@s - 1]) / 60) + " min ago; half of any move fades in " + (.halflife / 3600) + " h.";
		end;
	}
	dispbottom "Market: on, strength " + .strength + "%, half-life " + (.halflife / 3600) + " h, news " + (.news ? "on" : "off") + ", announcements " + (.announce ? "on" : "off") + ", " + $pv_mk_n + " item(s) moved, " + $pv_nw_n + " news.";
	for (.@i = 0; .@i < getarraysize($@pv_pub) && .@i < 15; .@i++) {
		.@id = $@pv_pub[.@i];
		.@pct = $@pop_item_pct[.@id];
		dispbottom "  " + getitemname(.@id) + " " + (.@pct >= 100 ? "+" : "") + (.@pct - 100) + "%";
	}
	for (.@i = 0; .@i < getarraysize($@pv_board_news$); .@i++)
		dispbottom "  news: " + $@pv_board_news$[.@i];
	dispbottom "@market <item> | news [<event>] | reset";
	end;
}
