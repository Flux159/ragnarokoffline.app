//===== rAthena Script =======================================
//= Buffer
//===== Description: =========================================
//= Free instant heal + buffs on click, no dialog window.
//===== How to customize: ====================================
//= - Duration of each buff: the number right before the last
//=   comma in each sc_start/sc_start4 line, in milliseconds
//=   (600000 = 10 min, 360000 = 6 min).
//= - Buff strength/level: the number right after the duration.
//= - To add/remove a buff: add/remove an sc_start line with the
//=   matching SC_ status and, if you want an animation, a
//=   specialeffect2 EF_ line above it.
//= - Soul Link is class-specific: the switch(BaseJob) block picks
//=   the right spirit skill ID for the clicking player's class.
//=   BaseJob has NO parentheses/argument -- it's a pseudo-variable,
//=   not a function. SL_HIGH (494) is the fallback for classes with
//=   no dedicated spirit. Spirit IDs are not confirmed against this
//=   server's own skill_db -- check with @ii <skill name> and fix
//=   the relevant case line if one is off.
//= - Sprite/name here only affects the (disabled) template; each
//=   duplicate() line below sets its own sprite and can be changed
//=   independently.
//============================================================
-	script	Buffer	-1,{
	specialeffect2 EF_HEAL2;
	percentheal 100,100;

	specialeffect2 EF_BLESSING;
	sc_start SC_BLESSING,600000,10;

	specialeffect2 EF_INCAGILITY;
	sc_start SC_INCREASEAGI,600000,10;

	specialeffect2 EF_ASSUMPTIO;
	sc_start SC_ASSUMPTIO,600000,5;

	specialeffect2 EF_KYRIE;
	sc_start SC_KYRIE,600000,5;

	specialeffect2 EF_MAGNIFICAT;
	sc_start SC_MAGNIFICAT,600000,5;

	// Soul Link, matched to the player's class; SL_HIGH as fallback.
	.@link = 494; // SL_HIGH
	switch(BaseJob) {
		case Job_Knight:         .@link = 452; break; // SL_KNIGHT
		case Job_Priest:         .@link = 454; break; // SL_PRIEST
		case Job_Wizard:         .@link = 453; break; // SL_WIZARD
		case Job_Blacksmith:     .@link = 458; break; // SL_BLACKSMITH
		case Job_Hunter:         .@link = 460; break; // SL_HUNTER
		case Job_Assassin:       .@link = 457; break; // SL_ASSASIN
		case Job_Crusader:       .@link = 450; break; // SL_CRUSADER
		case Job_Monk:           .@link = 447; break; // SL_MONK
		case Job_Sage:           .@link = 449; break; // SL_SAGE
		case Job_Rogue:          .@link = 456; break; // SL_ROGUE
		case Job_Alchemist:      .@link = 445; break; // SL_ALCHEMIST
		case Job_Bard:
		case Job_Dancer:         .@link = 455; break; // SL_BARDDANCER
		case Job_Star_Gladiator: .@link = 448; break; // SL_STAR
		case Job_Soul_Linker:    .@link = 461; break; // SL_SOULLINKER
		case Job_Super_Novice:   .@link = 451; break; // SL_SUPERNOVICE
	}
	sc_start4 SC_SPIRIT,600000,5,.@link,0,0; // val1=level, val2=spirit ID

	sc_start SC_KAUPE,360000,3;	// dodge chance on next few hits
	sc_start SC_KAAHI,360000,7;	// auto-heal below an HP threshold
	sc_start SC_KAIZEL,360000,7;	// auto-revive with partial HP on death

	dispbottom "Fully healed! Blessing, Agi Up, Assumptio, Kyrie, Magnificat, Soul Link, Kaupe, Kaahi and Kaizel for 10 minutes.";
	end;
}
